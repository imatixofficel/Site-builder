# ===== Create a file named .htaccess inside wp-content/uploads and put these lines in it =====
# It stops anyone from running a hidden script that was uploaded as an "image".
<IfModule mod_php.c>
php_flag engine off
</IfModule>
<IfModule mod_php7.c>
php_flag engine off
</IfModule>
<IfModule mod_php8.c>
php_flag engine off
</IfModule>
<FilesMatch "(?i)\.(php\d?|phtml|phar|pl|py|cgi|sh)$">
<IfModule mod_authz_core.c>
Require all denied
</IfModule>
<IfModule !mod_authz_core.c>
Order deny,allow
Deny from all
</IfModule>
</FilesMatch>
